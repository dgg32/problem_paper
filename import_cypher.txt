// replaced the old merged `Author` model with the instance-based `AuthorInstance`

// ---------------------------------------------------------------
//   1. Run THIS file (backbone). It loads Paper/Journal/Institution/Reason.
//   2. Run the author + identity pipeline (all in graph_processing/, idempotent,
//      in this order). These read data/graph/graph.json and write straight to
//      Neo4j over bolt; build_instances.py first deletes any old :Author /
//      :AuthorInstance nodes, so it is safe to re-run:
//        python graph_processing/build_instances.py    // AuthorInstance + WROTE + AFFILIATED_WITH

//        python graph_processing/link_instances.py       // PROBABLY_SAME_AS edges (+ MERGES from overrides)

// SETUP for step 1: LOAD CSV reads from the DBMS import dir. Copy the backbone

// Every load uses MERGE, so this script is idempotent. Fresh start:

// --- 1. Constraints (backbone; author-layer constraints are created by
//        build_instances.py) --------------------------------------------
CREATE CONSTRAINT paper_doi    IF NOT EXISTS FOR (p:Paper)       REQUIRE p.doi            IS UNIQUE;
CREATE CONSTRAINT journal_name IF NOT EXISTS FOR (j:Journal)     REQUIRE j.name           IS UNIQUE;
CREATE CONSTRAINT inst_id      IF NOT EXISTS FOR (i:Institution) REQUIRE i.institution_id IS UNIQUE;
CREATE CONSTRAINT reason_code  IF NOT EXISTS FOR (r:Reason)      REQUIRE r.code           IS UNIQUE;


// --- 2. Nodes ------------------------------------------------------

// Paper  (is_retracted -> boolean; dates -> Neo4j date type, empty -> null)
// TSV dates are ISO YYYY-MM-DD, which date() parses directly.
LOAD CSV WITH HEADERS FROM 'file:///nodes_paper.tsv' AS row FIELDTERMINATOR '\t'
MERGE (p:Paper {doi: row.doi})
SET p.pmid              = row.pmid,
    p.openalex_id       = row.openalex_id,
    p.title             = row.title,
    p.published_date    = CASE WHEN row.published_date  = '' THEN null ELSE date(row.published_date)  END,
    p.is_retracted      = toBoolean(row.is_retracted),
    p.retraction_date   = CASE WHEN row.retraction_date = '' THEN null ELSE date(row.retraction_date) END,
    p.retraction_nature = row.retraction_nature;

// Journal
LOAD CSV WITH HEADERS FROM 'file:///nodes_journal.tsv' AS row FIELDTERMINATOR '\t'
MERGE (j:Journal {name: row.name})
SET j.publisher = CASE WHEN row.publisher = '' THEN null ELSE row.publisher END;

// Institution
LOAD CSV WITH HEADERS FROM 'file:///nodes_institution.tsv' AS row FIELDTERMINATOR '\t'
MERGE (i:Institution {institution_id: row.institution_id})
SET i.name    = row.name,
    i.country = CASE WHEN row.country = '' THEN null ELSE row.country END;

// Reason
LOAD CSV WITH HEADERS FROM 'file:///nodes_reason.tsv' AS row FIELDTERMINATOR '\t'
MERGE (:Reason {code: row.code});


// --- 3. Relationships (paper-level; author edges come from the pipeline) --

// (Paper)-[:PUBLISHED_IN]->(Journal)
LOAD CSV WITH HEADERS FROM 'file:///rel_published_in.tsv' AS row FIELDTERMINATOR '\t'
MATCH (p:Paper   {doi: row.doi})
MATCH (j:Journal {name: row.journal_name})
MERGE (p)-[:PUBLISHED_IN]->(j);

// (Paper)-[:INVOLVES]->(Institution)  — paper-level institution pool (plan §1)
LOAD CSV WITH HEADERS FROM 'file:///rel_involves.tsv' AS row FIELDTERMINATOR '\t'
MATCH (p:Paper       {doi: row.doi})
MATCH (i:Institution {institution_id: row.institution_id})
MERGE (p)-[:INVOLVES]->(i);

// (Paper)-[:RETRACTED_FOR]->(Reason)
LOAD CSV WITH HEADERS FROM 'file:///rel_retracted_for.tsv' AS row FIELDTERMINATOR '\t'
MATCH (p:Paper  {doi: row.doi})
MATCH (r:Reason {code: row.reason_code})
MERGE (p)-[:RETRACTED_FOR]->(r);


// ===================================================================
// 4. Verification
//    Backbone (this file):  Paper 1554 | Journal 526 | Institution 2209 | Reason 92
//                           PUBLISHED_IN 1554 | INVOLVES 5208 | RETRACTED_FOR 5358
//    Author/identity layer (from the Python pipeline, not this file):
//                           AuthorInstance 10050 | WROTE 10050 | AFFILIATED_WITH 16880
//                           PROBABLY_SAME_AS 5133 | probable persons 7713 | adjudicated 201
// ===================================================================
MATCH (n) RETURN labels(n)[0] AS label, count(*) AS nodes ORDER BY nodes DESC;
